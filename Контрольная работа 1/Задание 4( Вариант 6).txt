#include <iostream>
#include <cmath>

using namespace std;

int main() {
    double a, b, c;
    char symbol;

    cout << "Введите a, b, c: ";
    cin >> a >> b >> c;

    cout << "Введите символ (F, o, j): ";
    cin >> symbol;

    if (symbol == 'F') {
        cout << "Ivan Ivanov" << endl;
    } 
    else if (symbol == 'o') {
        double d = b * b - 4 * a * c;
        if (d > 0) {
            cout << "x1 = " << (-b + sqrt(d)) / (2 * a) << endl;
            cout << "x2 = " << (-b - sqrt(d)) / (2 * a) << endl;
        } else if (d == 0) {
            cout << "x = " << -b / (2 * a) << endl;
        } else {
            cout << "Корней нет" << endl;
        }
    } 
    else if (symbol == 'j') {
        int num;
        cout << "Введите число: ";
        cin >> num;
        if (num % 2 == 0) {
            cout << "Делится на 2" << endl;
        } else {
            cout << "Не делится на 2" << endl;
        }
    } 
    else {
        cout << "Неверный символ" << endl;
    }

    return 0;
}
